```sparql
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX dig: <https://purl.org/okn/frink/kg/digcfdekg/schema/>
PREFIX wdt: <http://www.wikidata.org/prop/direct/>
# digcfdekg x prokn via Wikidata Entrez->ENSEMBL (P594) — the route reaching ProKN's FUNCTIONAL gene
# axis (up:Gene node = ensembl.org IRI), unlike the HGNC/P354 route (C13) which reaches only ~2% of it.
# CFDE type-2-diabetes genes (PIGEAN >= 7.0) each paired with ONE named diabetes-relevant ProKN
# MSigDB/Reactome pathway (RO_0002431) the gene carries.
SELECT ?sym ?pigeanScore (SAMPLE(?pwName) AS ?proknPathway) WHERE {
  GRAPH <https://purl.org/okn/frink/kg/digcfdekg> {
    ?st rdf:predicate dig:geneToTrait ; rdf:object <http://purl.obolibrary.org/obo/MONDO_0005148> ;
        rdf:subject ?gene ; dig:weight ?pigeanScore .
    ?gene rdfs:label ?sym . FILTER(?pigeanScore >= 7.0)
  }
  BIND(REPLACE(STR(?gene),'^.*/gene/','') AS ?entrez)
  GRAPH <https://purl.org/okn/frink/kg/wikidata> { ?item wdt:P351 ?entrez ; wdt:P594 ?ensg . }
  BIND(IRI(CONCAT('https://www.ensembl.org/id/', ?ensg)) AS ?pg)
  GRAPH <https://purl.org/okn/frink/kg/prokn> { ?pg <http://purl.obolibrary.org/obo/RO_0002431> ?pathway . }
  BIND(REPLACE(STR(?pathway),'^.*/rdf/MSigDB_','') AS ?pwName)
  FILTER(CONTAINS(?pwName,'INSULIN') || CONTAINS(?pwName,'GLUCOSE') || CONTAINS(?pwName,'DIABET')
      || CONTAINS(?pwName,'BETA_CELL') || CONTAINS(?pwName,'PANCREAT') || CONTAINS(?pwName,'LEPTIN')
      || CONTAINS(?pwName,'ZINC') || CONTAINS(?pwName,'GLUCAGON') || CONTAINS(?pwName,'ADIPO')
      || CONTAINS(?pwName,'INCRETIN') || CONTAINS(?pwName,'ENERGY_METAB') || CONTAINS(?pwName,'HNF'))
} GROUP BY ?sym ?pigeanScore ORDER BY DESC(?pigeanScore) LIMIT 15
```
